export const Resenas: React.FC<{ f: number; dur: number }> = ({ f, dur }) => {
  // el contador de reseñas y la nota suben durante toda la escena; arriba entra la de Nicolás,
  // abajo la de Carla reemplaza a la de Martín: siempre dos reseñas enteras, nada cortado
  const e1 = 18;
  const e2 = Math.max(e1 + 30, dur * 0.55);
  const entra1 = ease(f, e1, e1 + 18);
  const entra2 = ease(f, e2, e2 + 18);
  const nota = mixLin(f, e1, dur - 12, 4.6, 4.8);
  const cantidad = Math.round(mixLin(f, e1, dur - 12, 7, 9));
  const ap = ease(f, 0, 16);

  return (
    <div
      style={{
        position: 'absolute',
        left: 70,
        top: 330,
        width: 940,
        height: 920,
        borderRadius: 30,
        background: 'linear-gradient(180deg, #121B2E 0%, #0C1322 100%)',
        border: `1px solid ${CL_RES}`,
        boxShadow: '0 30px 70px rgba(0,0,0,0.55)',
        padding: '34px 36px',
        display: 'flex',
        flexDirection: 'column',
        gap: 20,
        overflow: 'hidden',
        boxSizing: 'border-box',
        fontFamily: sans,
        color: TXR,
        opacity: ap,
        transform: `translateY(${(1 - ap) * 40}px) scale(${0.968 + 0.032 * ap})`,
      }}
    >
      {/* encabezado de la tarjeta */}
      <div style={{ display: 'flex', alignItems: 'center', gap: 20 }}>
        <div style={{ width: 76, height: 76, borderRadius: 20, background: 'rgba(238,90,14,0.18)', display: 'flex', alignItems: 'center', justifyContent: 'center', flex: 'none' }}>
          <Ico d={D.estrella} size={40} color={NARANJA} fill />
        </div>
        <div style={{ minWidth: 0 }}>
          <div style={{ fontFamily: cond, fontSize: 44, fontWeight: 700, letterSpacing: '-0.02em', lineHeight: 1.08, color: TXR }}>
            Reseñas de Clientes Verificados
          </div>
          <div style={{ fontSize: 30, color: TXR3, marginTop: 2, letterSpacing: 1 }}>EXPERIENCIAS REALES</div>
        </div>
      </div>

      {/* 1 · la nota y el contador */}
      <div style={{ display: 'flex', alignItems: 'center', gap: 20 }}>
        <div
          style={{
            flex: 'none',
            padding: '10px 30px',
            borderRadius: 18,
            background: NARANJA,
            color: BLANCO,
            fontFamily: cond,
            fontWeight: 800,
            fontSize: 72,
            lineHeight: 1.1,
            letterSpacing: '-0.02em',
          }}
        >
          {nota.toFixed(1).replace('.', ',')}
        </div>
        <div style={{ minWidth: 0 }}>
          <EstrellasRes n={5} size={34} />
          <div style={{ fontSize: 32, color: TXR2, marginTop: 8, display: 'flex', alignItems: 'center', gap: 12 }}>
            <b style={{ fontFamily: cond, fontSize: 44, color: TXR }}>{cantidad}</b> reseñas verificadas
            <span style={{ width: 12, height: 12, borderRadius: 999, background: EXITO, opacity: 0.3 + 0.7 * Math.abs(Math.sin(f / 9)) }} />
          </div>
        </div>
      </div>

      {/* 2 · las barras por estrellas */}
      <div style={{ display: 'flex', flexDirection: 'column', gap: 12 }}>
        {[5, 4].map((e, i) => {
          const objetivo = e === 5 ? 1 : 0.3;
          const g = lin(f, e1 + i * 8, e1 + i * 8 + 26);
          return (
            <div key={e} style={{ display: 'flex', alignItems: 'center', gap: 18 }}>
              <div style={{ fontSize: 30, color: TXR3, width: 180, flex: 'none' }}>{e} estrellas</div>

                <div style={{ width: `${objetivo * g * 100}%`, height: '100%', background: ALERTA }} />
              </div>
            </div>
          );
        })}
      </div>

      {/* 3 · la reseña nueva de Nicolás, entera */}
      <div style={{ position: 'relative', height: 220, flex: 'none' }}>
        <div style={{ position: 'absolute', inset: 0 }}>
          <FilaResena
            nombre="Nicolás Romero"
            estrellas={5}
            texto="Me pasó el plano en 3D antes de cortar y la biblioteca entró justa contra el techo. Cobró como habíamos acordado en el chat, sin sorpresas."
            op={entra1}
            dy={(1 - entra1) * -34}
            alto={220}
            borde={NARANJA}
            fondo="rgba(238,90,14,0.10)"

          <FilaResena
            nombre="Carla Ferreyra"
            estrellas={5}
            texto="Vino el día que dijo, dejó todo limpio y el saldo lo pagué recién cuando terminó."
            op={1}
            alto={160}
            borde={EXITO}
            fondo="rgba(15,157,99,0.10)"
          />
        </div>
      </div>
    </div>
  );
};
