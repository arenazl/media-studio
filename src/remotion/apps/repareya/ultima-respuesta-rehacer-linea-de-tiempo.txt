// ── 9. mis solicitudes: la barra avanza tramo por tramo ──
export const LineaDeTiempo: React.FC<{ f: number; dur: number }> = ({ f, dur }) => {
  const arranque = 12;
  const paso = Math.max(20, (dur - arranque - 14) / 4);
  const avance = Math.max(0, (f - arranque) / paso);
  const etapa = Math.min(3, Math.floor(avance));
  const rotulos = ['Esperando presupuesto', 'Esperando tu respuesta', 'Trabajo agendado', 'Terminado'];
  const colores = [TXT3, INFO, NARANJA, EXITO];
  const tramos = ['Pedido', 'Presupuesto', 'Seña', 'Terminado'];
  const monto = etapa >= 1 ? pesos(Math.round(mixLin(f, arranque + paso, arranque + paso * 1.6, 0, 620000) / 1000) * 1000) : '—';
  const estado = etapa === 0 ? ['PENDIENTE', TXT3] : etapa === 1 ? ['COTIZADO', INFO] : etapa === 2 ? ['CONFIRMADO', NARANJA] : ['TERMINADO', EXITO];
  const llenado = (i: number) => lin(f, arranque + i * paso, arranque + i * paso + paso * 0.85);
  const comprometido = etapa >= 2 ? pesos(Math.round(mixLin(f, arranque + paso * 2, arranque + paso * 2.6, 0, 620000) / 1000) * 1000) : '$0';
  const pct = Math.round(Math.min(100, (avance / 4) * 100));

  const SUP = '#141C2B';
  const SUP2 = '#0E1522';
  const LINEA = 'rgba(255,255,255,0.10)';
  const BL = '#F4F6FB';
  const BL2 = 'rgba(244,246,251,0.70)';
  const BL3 = 'rgba(244,246,251,0.46)';

  const entra = (i: number) => ease(f, 10 + i * 7, 28 + i * 7);

  return (
    <div
      style={{
        position: 'absolute',
        left: 90,
        top: 330,
        width: 900,
        height: 920,
        borderRadius: 28,
        background: `linear-gradient(180deg, ${SUP} 0%, ${SUP2} 100%)`,
        border: `1px solid ${LINEA}`,
        boxShadow: '0 30px 70px rgba(0,0,0,0.55)',
        overflow: 'hidden',
        padding: '34px 36px',
        display: 'flex',
        flexDirection: 'column',
        gap: 24,
        fontFamily: sans,
        color: BL,
        opacity: ease(f, 0, 16),
        transform: `translateY(${(1 - ease(f, 0, 16)) * 40}px) scale(${0.97 + 0.03 * ease(f, 0, 16)})`,
      }}
    >
      {/* encabezado */}
      <div style={{ display: 'flex', alignItems: 'center', gap: 20 }}>
        <div style={{ width: 76, height: 76, borderRadius: 20, background: 'rgba(238,90,14,0.18)', display: 'flex', alignItems: 'center', justifyContent: 'center', flex: 'none' }}>
          <Ico d={D.bandeja} size={40} color={NARANJA} />
        </div>
        <div>
          <div style={{ fontFamily: cond, fontWeight: 800, fontSize: 44, lineHeight: 1.08, letterSpacing: '-0.02em', color: BL }}>Mis solicitudes</div>
          <div style={{ fontSize: 30, color: BL2, marginTop: 2 }}>Cotizaciones recibidas</div>
        </div>
      </div>

      {/* 1 · el trabajo, con su estado y su monto */}
      <div
        style={{
          padding: '26px 28px',
          borderRadius: 20,
          background: 'rgba(255,255,255,0.045)',
          border: `2px solid ${colores[etapa]}`,
          opacity: entra(0),
          transform: `translateY(${(1 - entra(0)) * 18}px)`,
        }}
      >
        <div style={{ display: 'flex', alignItems: 'flex-start', justifyContent: 'space-between', gap: 18 }}>
          <div>
            <div style={{ fontSize: 36, fontWeight: 700, lineHeight: 1.12 }}>Carpintería Marcelo</div>
            <div style={{ fontSize: 30, color: BL2, marginTop: 4 }}>Muebles a medida</div>
          </div>
          <Pastilla texto={estado[0] as string} color={estado[1] as string} size={30} fuerte />
        </div>
        <div style={{ fontFamily: cond, fontWeight: 800, fontSize: 86, lineHeight: 1.04, letterSpacing: '-0.03em', marginTop: 12, color: etapa >= 1 ? BL : BL3 }}>
          {monto}
        </div>
      </div>

      {/* 2 · los cuatro tramos */}
      <div style={{ opacity: entra(1), transform: `translateY(${(1 - entra(1)) * 18}px)` }}>
        <div style={{ display: 'flex', gap: 10 }}>
          {tramos.map((tr, i) => (
            <div key={i} style={{ flex: 1 }}>
              <div style={{ height: 18, borderRadius: 999, background: 'rgba(255,255,255,0.12)', overflow: 'hidden' }}>
                <div style={{ width: `${llenado(i) * 100}%`, height: '100%', background: i < etapa ? EXITO : NARANJA }} />
              </div>
              <div
                style={{
                  marginTop: 12,
                  fontSize: 30,
                  fontWeight: i === etapa ? 700 : 500,
                  color: llenado(i) > 0.4 ? BL : BL3,
                  textAlign: 'center',
                }}
              >
                {tr}
              </div>
            </div>
          ))}
        </div>
      </div>

      {/* 3 · el rótulo del paso, con el pulso vivo */}
      <div
        style={{
          display: 'flex',
          alignItems: 'center',
          gap: 14,
          padding: '20px 26px',
          borderRadius: 999,
          background: colores[etapa] + '24',
          color: colores[etapa],
          fontSize: 34,
          fontWeight: 700,
          opacity: entra(2),
          transform: `translateY(${(1 - entra(2)) * 18}px)`,
        }}
      >
        <Ico d={etapa === 3 ? D.check : etapa === 2 ? D.reloj : D.recibo} size={34} color={colores[etapa]} />
        <span>{rotulos[etapa]}</span>
        <span
          style={{
            marginLeft: 'auto',
            width: 16,
            height: 16,
            borderRadius: 999,
            background: colores[etapa],
            opacity: 0.25 + 0.75 * Math.abs(Math.sin(f / 7)),
            flex: 'none',
          }}
        />
      </div>

      {/* 4 · gasto comprometido y el avance general */}
      <div
        style={{
          marginTop: 'auto',
          padding: '24px 28px',
          borderRadius: 20,
          background: 'rgba(255,255,255,0.045)',
          border: `1px solid ${LINEA}`,
          opacity: entra(3),
          transform: `translateY(${(1 - entra(3)) * 18}px)`,
        }}
      >
        <div style={{ display: 'flex', alignItems: 'flex-end', justifyContent: 'space-between', gap: 18 }}>
          <div>
            <div style={{ fontSize: 30, letterSpacing: 1.5, color: BL3, textTransform: 'uppercase' }}>Gasto comprometido</div>
            <div style={{ fontFamily: cond, fontWeight: 800, fontSize: 64, lineHeight: 1.05, letterSpacing: '-0.02em', color: etapa >= 2 ? NARANJA : BL3, marginTop: 4 }}>
              {comprometido}
            </div>
          </div>
          <div style={{ fontFamily: cond, fontWeight: 800, fontSize: 56, letterSpacing: '-0.02em', color: EXITO }}>{pct}%</div>
        </div>
        <div style={{ marginTop: 18, height: 16, borderRadius: 999, background: 'rgba(255,255,255,0.12)', overflow: 'hidden' }}>
          <div style={{ width: `${pct}%`, height: '100%', background: EXITO }} />
        </div>
      </div>
    </div>
  );
};
