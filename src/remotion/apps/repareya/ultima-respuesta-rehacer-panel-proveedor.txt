  const TXT_N3 = 'rgba(247,241,234,0.46)';

  const entra = ease(f, 0, 16);

    { l: 'SOLICITUDES', v: mixLin(f, a, b, 12, 14), fmt: (n: number) => String(Math.round(n)), p: 'Todo respondido' },
    { l: 'CONFIRMADAS', v: mixLin(f, a, b, 9, 10), fmt: (n: number) => String(Math.round(n)), p: 'Últimos 30 días' },

    { l: 'CONVERSIÓN', v: mixLin(f, a, b, 75, 71), fmt: (n: number) => Math.round(n) + '%', p: '12 aceptadas de 14' },
  ];
  // barras del mes: crecen tramo por tramo a lo largo de toda la escena
  const barras = [0.42, 0.55, 0.48, 0.68, 0.74, 0.86, 1];
  const dias = ['L', 'M', 'M', 'J', 'V', 'S', 'D'];

  return (
    <div

        height: 920,
        borderRadius: 34,
        background: SUP,

        flexDirection: 'column',
        gap: 22,
        opacity: entra,

        color: TXT_N,
      }}

          </div>
          <div style={{ fontSize: 30, color: TXT_N2, marginTop: 4 }}>Tu performance de los últimos 30 días</div>
        </div>
      </div>

      <div style={{ display: 'grid', gridTemplateColumns: '1fr 1fr', gap: 18 }}>

            );
