// ── 1. contar el problema: el formulario se completa solo ──
const CP_SUP = '#19110C';
const CP_SUP2 = '#120C08';
const CP_LINEA = 'rgba(255,255,255,0.10)';
const CP_TXT = '#F6EFE8';
const CP_TXT2 = 'rgba(246,239,232,0.70)';
const CP_TXT3 = 'rgba(246,239,232,0.44)';

const CPFila: React.FC<{ etiqueta: string; valor: string; op?: number; dx?: number; grande?: boolean; borde?: string }> = ({
  etiqueta, valor, op = 1, dx = 0, grande = false, borde = CP_LINEA,
}) => (
  <div
    style={{
      display: 'flex',
      alignItems: 'center',
      justifyContent: 'space-between',
      gap: 20,
      padding: '26px 30px',
      borderRadius: 18,
      background: CP_SUP2,
      border: `1px solid ${borde}`,
      opacity: op,
      transform: `translateX(${dx}px)`,
    }}
  >
    <span style={{ fontSize: 32, color: CP_TXT3, letterSpacing: 1, textTransform: 'uppercase', fontWeight: 600 }}>{etiqueta}</span>
    <span style={{ fontSize: grande ? 56 : 36, fontWeight: 700, color: grande ? NARANJA : CP_TXT, fontFamily: grande ? cond : sans, letterSpacing: grande ? '-0.02em' : 0 }}>{valor}</span>
  </div>
);

export const ContarProblema: React.FC<{ f: number; dur: number }> = ({ f, dur }) => {
  const t = Math.max(30, (dur - 40) / 3);
  const etapa = f < 14 + t ? 0 : f < 14 + t * 2 ? 1 : 2;
  const elige = ease(f, 18, 34);
  const texto = 'Se me llueve el techo del lavadero cuando llueve fuerte...';
  const tipeo = texto.slice(0, Math.round(lin(f, 14 + t + 6, 14 + t + t * 0.8) * texto.length));
  const publicando = f > 14 + t * 2 + 26;
  const activo = etapa === 0 ? 0 : etapa === 1 ? 2 : 3;
  const pasos = ['Tipo', 'Detalles', 'Lugar', 'Resumen'];
  const ap = ease(f, 0, 16);

  return (
    <div
      style={{
        position: 'absolute',
        left: 75,
        top: 330,
        width: 930,
        height: 920,
        borderRadius: 34,
        background: `linear-gradient(180deg, ${CP_SUP} 0%, ${CP_SUP2} 100%)`,
        border: `1px solid ${CP_LINEA}`,
        boxShadow: '0 34px 80px rgba(0,0,0,0.55)',
        overflow: 'hidden',
        opacity: ap,
        transform: `translateY(${(1 - ap) * 40}px) scale(${0.97 + 0.03 * ap})`,
        fontFamily: sans,
        color: CP_TXT,
        padding: '38px 40px',
        display: 'flex',
        flexDirection: 'column',
        gap: 28,
      }}
    >
      {/* encabezado */}
      <div style={{ display: 'flex', alignItems: 'center', gap: 22 }}>
        <div style={{ width: 86, height: 86, borderRadius: 22, background: 'rgba(238,90,14,0.18)', display: 'flex', alignItems: 'center', justifyContent: 'center', flex: 'none' }}>
          <Ico d={D.llave} size={46} color={NARANJA} />
        </div>
        <div>
          <div style={{ fontFamily: cond, fontSize: 44, fontWeight: 700, letterSpacing: '-0.02em', lineHeight: 1.08, color: CP_TXT }}>
            Contanos qué hay que arreglar
          </div>
          <div style={{ fontSize: 30, color: CP_TXT2, marginTop: 4 }}>{`Paso ${activo + 1} de 4`}</div>
        </div>
      </div>

      {/* barra de pasos */}
      <div style={{ display: 'flex', gap: 12 }}>
        {pasos.map((p, i) => (
          <div key={i} style={{ flex: 1 }}>
            <div style={{ height: 10, borderRadius: 999, background: 'rgba(255,255,255,0.12)', overflow: 'hidden' }}>
              <div style={{ width: `${i <= activo ? 100 : 0}%`, height: '100%', background: i < activo ? EXITO : NARANJA }} />
            </div>
            <div style={{ marginTop: 12, fontSize: 30, fontWeight: i === activo ? 700 : 500, color: i <= activo ? CP_TXT : CP_TXT3 }}>{p}</div>
          </div>
        ))}
      </div>

      {/* cuerpo: tres estados, mismo lugar reservado */}
      <div style={{ position: 'relative', flex: 1 }}>
        {etapa === 0 ? (
          <div style={{ display: 'flex', flexDirection: 'column', gap: 18, height: '100%' }}>
            {['Humedad y filtraciones', 'Electricidad', 'Plomería', 'Pintura'].map((t2, i) => {
              const on = i === 0 && elige > 0.5;
              return (
                <div
                  key={i}
                  style={{
                    flex: 1,
                    padding: '0 30px',
                    borderRadius: 20,
                    background: on ? 'rgba(238,90,14,0.16)' : CP_SUP2,
                    border: `2px solid ${on ? NARANJA : CP_LINEA}`,
                    display: 'flex',
                    alignItems: 'center',
                    gap: 22,
                    transform: `scale(${on ? 1 + 0.02 * elige : 1})`,
                  }}
                >
                  <Ico d={i === 0 ? D.gota : D.llave} size={40} color={on ? NARANJA : CP_TXT3} />
                  <span style={{ fontSize: 38, fontWeight: on ? 700 : 500, color: on ? NARANJA : CP_TXT2 }}>{t2}</span>
                  {on ? <div style={{ marginLeft: 'auto' }}><Ico d={D.check} size={38} color={NARANJA} stroke={3} /></div> : null}
                </div>
              );
            })}
          </div>
        ) : etapa === 1 ? (
          <div style={{ display: 'flex', flexDirection: 'column', gap: 18, height: '100%' }}>
            <CPFila etiqueta="Zona" valor="Villa Urquiza" />
            <CPFila etiqueta="Tipo de trabajo" valor="Humedad y filtraciones" />
            <div
              style={{
                flex: 1,
                padding: '26px 30px',
                borderRadius: 20,
                background: CP_SUP2,
                border: `2px solid ${NARANJA}`,
                display: 'flex',
                flexDirection: 'column',
              }}
            >
              <div style={{ fontSize: 30, color: CP_TXT3, letterSpacing: 1, textTransform: 'uppercase', fontWeight: 600 }}>
                Título del trabajo
              </div>
              <div style={{ fontSize: 34, lineHeight: 1.35, color: CP_TXT, marginTop: 10 }}>
                {tipeo}
                <span style={{ opacity: Math.sin(f / 3) > 0 ? 1 : 0, color: NARANJA }}>|</span>
              </div>
              <div style={{ display: 'flex', justifyContent: 'flex-end', marginTop: 'auto' }}>
                <div
                  style={{
                    width: 76,
                    height: 76,
                    borderRadius: 999,
                    background: 'rgba(238,90,14,0.18)',
                    display: 'flex',
                    alignItems: 'center',
                    justifyContent: 'center',
                    transform: `scale(${1 + 0.08 * Math.abs(Math.sin(f / 8))})`,
                  }}
                >
                  <Ico d={D.mic} size={40} color={NARANJA} />
                </div>
              </div>
            </div>
          </div>
        ) : (
          <div style={{ display: 'flex', flexDirection: 'column', gap: 18, height: '100%' }}>
            {[
              ['Tipo de trabajo', 'Humedad y filtraciones', false],
              ['Superficie', '12 m2', false],
              ['Presupuesto máximo', '$400.000', true],
            ].map((r, i) => {
              const a = ease(f, 14 + t * 2 + i * 6, 14 + t * 2 + i * 6 + 14);
              return (
                <CPFila
                  key={i}
                  etiqueta={r[0] as string}
                  valor={r[1] as string}
                  grande={r[2] as boolean}
                  op={a}
                  dx={(1 - a) * 20}
                />
              );
            })}
            <div
              style={{
                marginTop: 'auto',
                height: 108,
                borderRadius: 20,
                background: publicando ? NARANJA_OSCURO : NARANJA,
                color: BLANCO,
                fontFamily: cond,
                fontSize: 44,
                fontWeight: 700,
                letterSpacing: '-0.01em',
                display: 'flex',
                alignItems: 'center',
                justifyContent: 'center',
                boxShadow: '0 14px 34px rgba(238,90,14,0.35)',
                transform: `scale(${publicando ? 0.985 : 1 + 0.015 * Math.abs(Math.sin(f / 7))})`,
              }}
            >
              {publicando ? 'Publicando…' : 'Crear solicitud'}
            </div>
          </div>
        )}
      </div>
    </div>
  );
};
