// ── 6. plan de pagos: del aviso rojo al verde, etapa por etapa ──
const CARD_X = 72;
const CARD_W = 936;
const CARD_TOP = 330;
const CARD_H = 920;
const PANEL = '#121B2E';
const PANEL_2 = '#0C1322';
const LINEA_PAGOS = 'rgba(255,255,255,0.10)';
const T_CLARO = '#F4F6FB';
const T_MEDIO = 'rgba(255,255,255,0.68)';
const T_TENUE = 'rgba(255,255,255,0.46)';
const VERDE_N = '#3DD68C';
const ROJO_N = '#F87171';

const TarjetaOscura: React.FC<{ f: number; titulo: string; sub: string; icono: string; children: React.ReactNode }> = ({ f, titulo, sub, icono, children }) => {
  const a = ease(f, 0, 18);
  return (
    <div
      style={{
        position: 'absolute',
        left: CARD_X,
        top: CARD_TOP,
        width: CARD_W,
        height: CARD_H,
        borderRadius: 32,
        background: `linear-gradient(180deg, ${PANEL} 0%, ${PANEL_2} 100%)`,
        border: `1px solid ${LINEA_PAGOS}`,
        boxShadow: '0 36px 90px rgba(0,0,0,0.58)',
        overflow: 'hidden',
        padding: '38px 40px 40px',
        display: 'flex',
        flexDirection: 'column',
        gap: 24,
        fontFamily: sans,
        color: T_CLARO,
        opacity: a,
        transform: `translateY(${(1 - a) * 40}px) scale(${0.968 + 0.032 * a})`,
      }}
    >
      <div style={{ display: 'flex', alignItems: 'center', gap: 22 }}>
        <div style={{ width: 84, height: 84, borderRadius: 22, background: 'rgba(238,90,14,0.18)', display: 'flex', alignItems: 'center', justifyContent: 'center', flex: 'none' }}>
          <Ico d={icono} size={44} color={NARANJA} />
        </div>
        <div style={{ minWidth: 0 }}>
          <div style={{ fontFamily: cond, fontWeight: 700, fontSize: 44, lineHeight: 1.08, letterSpacing: '-0.02em', color: T_CLARO }}>{titulo}</div>
          <div style={{ fontSize: 30, color: T_MEDIO, marginTop: 4 }}>{sub}</div>
        </div>
      </div>
      {children}
    </div>
  );
};

export const PlanDePagos: React.FC<{ f: number; dur: number }> = ({ f, dur }) => {
  // tres estados repartidos: todo al final (rojo) → mitad y mitad (rojo) → contra avance (verde)
  const t = Math.max(26, (dur - 30) / 3);
  const c1 = 16;            // elige "mitad y mitad"
  const c2 = 16 + t;        // elige "contra avance"
  const c3 = 16 + t * 2;    // habilita mandar propuesta
  const elegida = f < c1 ? 0 : f < c2 ? 1 : 2;
  const avance = f >= c2;
  const listo = f >= c3;
  const sw = ease(f, c3, c3 + 14);
  const plantillas = ['Todo al final', 'Mitad y mitad', 'Contra avance'];

  const pagos =
    elegida === 0
      ? [['1', 'Al terminar todo', '$620.000']]
      : elegida === 1
      ? [['1', 'Arranque', '$310.000'], ['2', 'Entrega', '$310.000']]
      : [['1', 'Materiales y arranque', '$124.000'], ['2', 'Avance de obra', '$248.000'], ['3', 'Saldo final', '$248.000']];

  // el primer pago baja de 100% a 50% y a 20%: el medidor se mueve toda la escena
  const primerPct = elegida === 0 ? mixLin(f, 10, c1, 100, 100) : elegida === 1 ? mixLin(f, c1, c1 + 14, 100, 50) : mixLin(f, c2, c2 + 18, 50, 20);
  const tope = 40;
  const base = elegida === 0 ? 10 : elegida === 1 ? c1 : c2;
  const anchoCol = (CARD_W - 80 - 24) / 3;
  const centro = (i: number) => 40 + anchoCol * i + anchoCol / 2 + 12 * i;
  const cursorX = elegida === 0 ? mixLin(f, 8, c1, centro(0) - 90, centro(0)) : elegida === 1 ? mixLin(f, c1, c2, centro(0), centro(1)) : mixLin(f, c2, c2 + 16, centro(1), centro(2));
  const apreta = interpolate(f % Math.max(1, Math.round(t)), [0, 4, 10], [0, 1, 0], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' }) * (listo ? 0 : 1);
  const fuerte = avance ? VERDE_N : ROJO_N;

  // el aviso corto (verde) ocupa una sola linea: el bloque de etapas y el de aviso tienen
  // alturas fijas que suman lo mismo en los tres estados, asi nada se monta ni se corta
  const ALTO_AVISO = avance ? 92 : 142;
  const ALTO_ETAPAS = 328 - ALTO_AVISO + 92; // 328 fijo entre etapas + aviso

  return (
    <TarjetaOscura f={f} titulo="¿Cómo cobrás este trabajo?" sub="Biblioteca a medida · $620.000" icono={D.recibo}>
      {/* plantillas: el cursor pasa de una a otra */}
      <div style={{ position: 'relative' }}>
        <div style={{ display: 'flex', gap: 12 }}>
          {plantillas.map((p, i) => {
            const on = i === elegida;
            return (
              <div
                key={i}
                style={{
                  flex: 1,
                  padding: '22px 10px',
                  borderRadius: 18,
                  background: on ? 'rgba(238,90,14,0.16)' : 'rgba(255,255,255,0.04)',
                  border: `2px solid ${on ? NARANJA : LINEA_PAGOS}`,
                  textAlign: 'center',
                  transform: `scale(${on ? 1.03 : 1})`,
                }}
              >
                <div style={{ fontSize: 30, fontWeight: 700, color: on ? NARANJA : T_MEDIO, lineHeight: 1.12 }}>{p}</div>
              </div>
            );
          })}
        </div>
        <Cursor x={cursorX} y={96} apreta={apreta} op={listo ? 0 : 1} />
      </div>

      {/* medidor del primer pago */}
      <div style={{ padding: '24px 28px', borderRadius: 20, background: 'rgba(255,255,255,0.04)', border: `1px solid ${LINEA_PAGOS}` }}>
        <div style={{ display: 'flex', alignItems: 'baseline', justifyContent: 'space-between' }}>
          <div style={{ fontSize: 30, color: T_TENUE, letterSpacing: 2, textTransform: 'uppercase', fontWeight: 600 }}>Primer pago</div>
          <div style={{ display: 'flex', alignItems: 'baseline', gap: 16 }}>
            <div style={{ fontFamily: cond, fontWeight: 800, fontSize: 72, lineHeight: 1, letterSpacing: '-0.02em', color: fuerte }}>
              {Math.round(primerPct)}%
            </div>
            <div style={{ fontSize: 30, color: T_MEDIO }}>tope {tope}%</div>
          </div>
        </div>
        <div style={{ marginTop: 18, position: 'relative', height: 20, borderRadius: 999, background: 'rgba(255,255,255,0.10)', overflow: 'hidden' }}>
          <div style={{ width: `${Math.min(100, primerPct)}%`, height: '100%', background: fuerte }} />
          <div style={{ position: 'absolute', left: `${tope}%`, top: 0, width: 4, height: 20, background: T_CLARO }} />
        </div>
      </div>

      {/* etapas: entran de a una. Lugar fijo, sin desbordar */}
      <div style={{ display: 'flex', flexDirection: 'column', gap: 14, height: ALTO_ETAPAS, justifyContent: 'flex-start', overflow: 'hidden' }}>
        {pagos.map((p, i) => {
          const a = ease(f, base + i * 6, base + i * 6 + 14);
          return (
            <div
              key={`${elegida}-${i}`}
              style={{
                display: 'flex',
                alignItems: 'center',
                gap: 20,
                padding: '18px 24px',
                borderRadius: 18,
                background: 'rgba(255,255,255,0.04)',
                border: `1px solid ${LINEA_PAGOS}`,
                flex: 'none',
                opacity: a,
                transform: `translateX(${(1 - a) * 24}px)`,
              }}
            >
              <div style={{ width: 58, height: 58, borderRadius: 999, background: 'rgba(238,90,14,0.18)', color: NARANJA, display: 'flex', alignItems: 'center', justifyContent: 'center', fontFamily: cond, fontWeight: 800, fontSize: 34, flex: 'none' }}>
                {p[0]}
              </div>
              <div style={{ flex: 1, fontSize: 32, fontWeight: 600, color: T_CLARO, lineHeight: 1.15 }}>{p[1]}</div>
              <div style={{ fontFamily: cond, fontWeight: 800, fontSize: 56, letterSpacing: '-0.02em', color: T_CLARO }}>{p[2]}</div>
            </div>
          );
        })}
      </div>

      {/* aviso: rojo hasta que entra en el tope. Tiene su propio lugar reservado */}
      <div
        style={{
          display: 'flex',
          alignItems: 'center',
          gap: 16,
          padding: '0 26px',
          borderRadius: 18,
          background: avance ? 'rgba(61,214,140,0.12)' : 'rgba(248,113,113,0.12)',
          border: `1px solid ${avance ? VERDE_N : ROJO_N}`,
          height: ALTO_AVISO,
          flex: 'none',
        }}
      >
        <div style={{ flex: 'none', display: 'flex', alignItems: 'center' }}>
          <Ico d={avance ? D.check : D.alerta} size={34} color={fuerte} />
        </div>
        <span style={{ fontSize: 32, lineHeight: 1.3, fontWeight: 600, color: fuerte }}>
          {avance
            ? 'Primer pago 20%. Podés pedir hasta 40%.'
            : 'El primer pago es el 50% del total y el máximo para vos es 40%. Bajá el primer pago o repartilo en más etapas.'}
        </span>
      </div>

      {/* botón: se habilita al final. Altura fija, siempre dentro de la tarjeta */}
      <div
        style={{
          height: 96,
          flex: 'none',
          borderRadius: 20,
          background: listo ? NARANJA : 'rgba(255,255,255,0.07)',
          color: listo ? BLANCO : T_TENUE,
          fontSize: 36,
          fontWeight: 700,
          display: 'flex',
          alignItems: 'center',
          justifyContent: 'center',
          gap: 14,
          border: listo ? 'none' : `1px solid ${LINEA_PAGOS}`,
          opacity: 0.65 + 0.35 * sw,
          transform: `scale(${listo ? 1 + 0.015 * Math.abs(Math.sin(f / 7)) : 1})`,
          boxShadow: listo ? '0 14px 36px rgba(238,90,14,0.38)' : 'none',
        }}
      >
        {listo ? <Ico d={D.check} size={34} color={BLANCO} /> : null}
        Mandar propuesta
      </div>
    </TarjetaOscura>
  );
};
