// Alto real de cada tarjeta (ya multiplicado por su escala), medido del render.
// De acá sale el top del título: borde inferior + AIRE, nunca encima de la tarjeta.
const ALTO_TARJETA: Record<string, number> = {
  'contar-problema': 735,
  'presupuestos-que-llegan': 600,
  'chat-plan-de-pagos': 740,
  'pagar-la-sena': 700,
  'bandeja-solicitudes': 745,
  'armar-plan-de-pagos': 748,
  'panel-proveedor': 650,
  'reputacion-resenas': 600,
  'mis-solicitudes': 700,
};

const topTitulo = (tipo: string) => {
  const alto = Math.min(TARJETA_ALTO_MAX, ALTO_TARJETA[tipo] ?? 700);
  return Math.max(TITULO_TOP_MIN, TARJETA_TOP + alto + AIRE);
};

const CuerpoEscena: React.FC<{ e: EscenaTarjetaApp & { dur: number } }> = ({ e }) => {
  const f = useCurrentFrame();
  const salida = 1 - lin(f, e.dur - 8, e.dur);
  const conocido = Object.prototype.hasOwnProperty.call(ALTO_TARJETA, e.tipo);
  if (!conocido) {
    return <AbsoluteFill style={{ opacity: salida }}><Titulo f={f} chip={e.chip} titulo={e.titulo} resaltar={e.resaltar} sub={e.sub} top={760} size={104} /></AbsoluteFill>;
  }
  return (
    <AbsoluteFill style={{ opacity: salida }}>
      {e.tipo === 'contar-problema' ? <ContarProblema f={f} dur={e.dur} /> : null}
      {e.tipo === 'presupuestos-que-llegan' ? <Esperan f={f} dur={e.dur} /> : null}
      {e.tipo === 'chat-plan-de-pagos' ? <ChatPlan f={f} dur={e.dur} /> : null}
      {e.tipo === 'pagar-la-sena' ? <PagarSena f={f} dur={e.dur} /> : null}
      {e.tipo === 'bandeja-solicitudes' ? <Bandeja f={f} dur={e.dur} /> : null}
      {e.tipo === 'armar-plan-de-pagos' ? <PlanDePagos f={f} dur={e.dur} /> : null}
      {e.tipo === 'panel-proveedor' ? <Panel f={f} dur={e.dur} /> : null}
      {e.tipo === 'reputacion-resenas' ? <Reputacion f={f} dur={e.dur} /> : null}
      {e.tipo === 'mis-solicitudes' ? <MisSolicitudes f={f} dur={e.dur} /> : null}
      <Titulo f={f} chip={e.chip} titulo={e.titulo} resaltar={e.resaltar} sub={e.sub} top={topTitulo(e.tipo)} size={74} />
    </AbsoluteFill>
  );
};

export const Tarjetas: React.FC<{ plan: PlanTarjetasApp }> = ({ plan }) => {
  const { fps } = useVideoConfig();
  const esc = useMemo(() => {
    let c = 0;
    return plan.escenas.map((e) => { const from = c; const dur = Math.round(e.durSec * fps); c += dur; return { ...e, from, dur }; });
  }, [plan, fps]);
  const finEscenas = esc.length ? esc[esc.length - 1].from + esc[esc.length - 1].dur : 0;
  const placaDur = Math.round(plan.placa.durSec * fps);
  return (
    <AbsoluteFill style={{ backgroundColor: MADERA_PP }}>
      <Fondo />
      <Sequence from={0} durationInFrames={finEscenas} name="marco">
        <Progreso escenas={esc} />
        <Logo src={plan.logoSrc} />
      </Sequence>
      {esc.map((e, i) => (
        <Sequence key={i} from={e.from} durationInFrames={e.dur} name={`${i + 1} ${e.tipo}`}>
          <CuerpoEscena e={e} />
        </Sequence>
      ))}
      {placaDur > 0 ? (
        <Sequence from={finEscenas} durationInFrames={placaDur} name="placa">
          <Placa plan={plan} />
        </Sequence>
      ) : null}
      <div style={{ position: 'absolute', bottom: 70, left: 0, right: 0, textAlign: 'center', fontFamily: saira, fontWeight: 600, fontSize: 25, color: 'rgba(255,255,255,0.72)', letterSpacing: 3 }}>reparehoy.com.ar</div>
    </AbsoluteFill>
  );
};
