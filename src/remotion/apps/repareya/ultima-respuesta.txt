export const Tarjetas: React.FC<{ plan: PlanTarjetasApp }> = ({ plan }) => {
  const { fps } = useVideoConfig();
  const esc = useMemo(() => {
    let c = 0;
    return plan.escenas.map((e) => {
      const from = c;
      const dur = Math.round(e.durSec * fps);
      c += dur;
      return { ...e, from, dur };
    });
  }, [plan, fps]);
  const fin = esc.length ? esc[esc.length - 1].from + esc[esc.length - 1].dur : 0;
  const placaDur = Math.round(plan.placa.durSec * fps);
  return (
    <AbsoluteFill style={{ backgroundColor: FONDO }}>
      <Fondo />
      <Sequence from={0} durationInFrames={fin} name="marco">
        <Progreso escenas={esc} />
        <Encabezado src={plan.logoSrc} />
        <Pie />
      </Sequence>
      {esc.map((e, i) => (
        <Sequence key={i} from={e.from} durationInFrames={e.dur} name={`${i + 1} ${e.tipo}`}>
          <Escena e={e} />
        </Sequence>
      ))}
      {placaDur > 0 ? (
        <Sequence from={fin} durationInFrames={placaDur} name="placa">
          <Placa plan={plan} />
        </Sequence>
      ) : null}
    </AbsoluteFill>
  );
};
